function sendVocabularyEmail() {
  const email = 'tienganhthanh06@gmail.com';
  const props = PropertiesService.getScriptProperties();
  let currentIndex = parseInt(props.getProperty('currentIndex')) || 0;
  
  if (currentIndex >= allData.length) currentIndex = 0;
  
  const words = allData.slice(currentIndex, currentIndex + 5);
  if (words.length === 0) return;
  
  let group = String.fromCharCode(65 + Math.floor(currentIndex / 50));
  if (group > 'J') group = 'J';
  
  let html = `<div style='font-family:sans-serif;max-width:500px;margin:auto;padding:20px;border-radius:12px;background:#f8fafc;border:1px solid #e2e8f0;'>
    <h2 style='text-align:center;color:#3b82f6;'>📚 5 Từ Vựng Hôm Nay</h2>
    <h4 style='text-align:center;color:#64748b;'>(Nhóm ${group})</h4>
    <div style='background:white;padding:15px;border-radius:8px;'>`;
    
  words.forEach((w, i) => {
    html += `<div style='margin-bottom:12px;border-bottom:1px solid #f1f5f9;padding-bottom:10px;'>
      <b style='font-size:18px;'>${currentIndex + i + 1}. ${w[0]}</b><br>
      <span style='color:#10b981;font-size:16px;'>👉 ${w[1]}</span>
    </div>`;
  });
  
  html += `</div>
    <p style='text-align:center;color:#94a2b8;'>Chúc bạn học tốt!</p>
  </div>`;
  
  MailApp.sendEmail({
    to: email, 
    subject: `[Nhóm ${group}] 📚 5 Từ Vựng Giao Tiếp Mỗi Ngày`, 
    htmlBody: html
  });
  
  props.setProperty('currentIndex', (currentIndex + 5).toString());
}

function setupDailyTriggers() {
  // Xóa các trigger cũ nếu có
  ScriptApp.getProjectTriggers().forEach(t => ScriptApp.deleteTrigger(t));
  
  // Hẹn giờ 10h sáng (Test)
  ScriptApp.newTrigger('sendVocabularyEmail').timeBased().everyDays(1).atHour(6).create();
  
  // Hẹn giờ 12h trưa
  ScriptApp.newTrigger('sendVocabularyEmail').timeBased().everyDays(1).atHour(12).create();
  
  // Hẹn giờ 19h tối (7 PM)
  ScriptApp.newTrigger('sendVocabularyEmail').timeBased().everyDays(1).atHour(19).create();
  
  // Chạy thử 1 lần đầu tiên ngay lập tức
  sendVocabularyEmail();
}
